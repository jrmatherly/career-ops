// Copy to web/tests/lib/ and run from web/: node --test tests/lib/<name>. Fails at 2edd1b33 with 'Caf�� ...'.
import { test } from "node:test";
import assert from "node:assert/strict";
import { EventEmitter } from "node:events";
import { StringDecoder } from "node:string_decoder";
import { markTrackerReady } from "../../src/lib/pdf-render.mjs";

function fakeStream() {
  const stream = new EventEmitter();
  let decoder = null;
  stream.setEncoding = (enc) => { decoder = new StringDecoder(enc); };
  stream.push = (buf) => { stream.emit("data", decoder ? decoder.write(buf) : buf); };
  return stream;
}

test("markTrackerReady decodes a multibyte character split across two stdout chunks", async () => {
  const bytes = Buffer.from(JSON.stringify({ ok: true, company: "Café Ünïcode 日本" }));
  const cut = bytes.indexOf(0xc3) + 1; // inside the two bytes of "é"
  const spawnFn = () => {
    const child = new EventEmitter();
    child.stdout = fakeStream();
    child.stderr = fakeStream();
    queueMicrotask(() => {
      child.stdout.push(bytes.subarray(0, cut));
      child.stdout.push(bytes.subarray(cut));
      child.emit("close", 0);
    });
    return child;
  };
  const result = await markTrackerReady({ spawnFn, execPath: "node", coreRoot: "/core", reportNum: "001" });
  assert.equal(result.data?.company, "Café Ünïcode 日本");
});
