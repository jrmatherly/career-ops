// Run from the repo root: node effects.mjs web/src. Lists useEffect blocks that fetch without a cancel guard or an error state (heuristic).
import fs from 'node:fs'; import path from 'node:path';
const root = process.argv[2]; const out=[];
const walk=d=>{for(const e of fs.readdirSync(d,{withFileTypes:true})){const p=path.join(d,e.name); if(e.isDirectory()) walk(p); else if(p.endsWith('.tsx')) scan(p);}};
function scan(f){const s=fs.readFileSync(f,'utf8'); let i=0; while((i=s.indexOf('useEffect(',i))!==-1){ // find matching paren
 let depth=0,j=i+9; for(;j<s.length;j++){const c=s[j]; if(c==='(')depth++; else if(c===')'){depth--; if(depth===0)break;}}
 const body=s.slice(i,j+1); const line=s.slice(0,i).split('\n').length;
 if(/fetch\(|fetchJson|getJson|loadJson/.test(body)){
  const guarded=/cancel|ignore|abort|alive|active|mounted|stale|disposed|current\s*!==|signal/i.test(body);
  const errState=/catch|\.ok|setError|error/i.test(body);
  if(!guarded||!errState) out.push(`${path.relative(root,f)}:${line} guarded=${guarded} err=${errState}`);
 } i=j;}}
walk(root); console.log(out.join('\n')); console.log('count',out.length);
